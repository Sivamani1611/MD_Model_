# ───────────────────────────────────────────────────────────────────────────────
# Load Model & Feature List
# ───────────────────────────────────────────────────────────────────────────────
@st.cache_resource
def load_artifacts():
    model = joblib.load("model/model.pkl")
    feature_list = pickle.load(open("model/features.pkl", "rb"))
    return model, feature_list

model, feature_list = load_artifacts()

# ───────────────────────────────────────────────────────────────────────────────
# Initialize Session Logs
# ───────────────────────────────────────────────────────────────────────────────
if "logs" not in st.session_state:
    st.session_state["logs"] = []

# ───────────────────────────────────────────────────────────────────────────────
# Main Tabs
# ───────────────────────────────────────────────────────────────────────────────
tab_scan, tab_features, tab_dashboard, tab_reports, tab_about = st.tabs([
    "🔬 File Scanner", "🔍 Feature Explorer", "📈 Dashboard", "📝 Reports", "ℹ️ About"
])

# ───────────────────────────────────────────────────────────────────────────────
# Tab 1: File Scanner
# ───────────────────────────────────────────────────────────────────────────────
with tab_scan:
    st.header("🔬 File Scanner")
    files = st.file_uploader(
        "Upload Windows PE files (exe, dll):", type=["exe", "dll"], accept_multiple_files=True
    )
    if files:
        for f in files:
            suffix = os.path.splitext(f.name)[1]
            tmp = tempfile.NamedTemporaryFile(delete=False, suffix=suffix)
            tmp.write(f.getvalue()); tmp.close()

            file_hash = hashlib.sha256(f.getvalue()).hexdigest()
            features = extract_info(tmp.name)
            os.remove(tmp.name)
            if not features:
                st.error(f"Invalid PE file: {f.name}")
                continue

            df = pd.DataFrame([features])[feature_list]
            pred = model.predict(df)[0]
            label = "LEGITIMATE" if pred else "MALWARE"
            css = "legit" if pred else "malw"

            st.session_state["logs"].append({
                "time": time.time(),
                "file": f.name,
                "hash": file_hash,
                "label": label,
                **features
            })

            col1, col2 = st.columns([4,1])
            col1.markdown(
                f"**{f.name}** — <span class='{css} result-label'>{label}</span>",
                unsafe_allow_html=True
            )
            col2.code(file_hash[:16] + "…")

# ───────────────────────────────────────────────────────────────────────────────
# Tab 2: Feature Explorer
# ───────────────────────────────────────────────────────────────────────────────
with tab_features:
    st.header("🔍 Feature Explorer")
    logs = st.session_state["logs"]
    if not logs:
        st.info("No files scanned yet. Use the File Scanner tab.")
    else:
        names = [e["file"] for e in logs]
        sel = st.selectbox("Select a scanned file:", names)
        ent = next(e for e in logs if e["file"] == sel)

        df_feat = pd.DataFrame.from_dict({k: ent[k] for k in feature_list}, orient='index', columns=['Value'])
        df_feat.index.name = 'Feature'
        st.dataframe(df_feat, use_container_width=True)
        csv = df_feat.to_csv().encode()
        st.download_button("📥 Download features as CSV", csv, file_name=f"{sel}_features.csv")

# ───────────────────────────────────────────────────────────────────────────────
# Tab 3: Dashboard
# ───────────────────────────────────────────────────────────────────────────────
with tab_dashboard:
    st.header("📈 Dashboard & Trends")
    df_logs = pd.DataFrame(st.session_state["logs"])
    if df_logs.empty:
        st.info("No scans performed yet.")
    else:
        df_logs['datetime'] = pd.to_datetime(df_logs['time'], unit='s')
        df_logs['date'] = df_logs['datetime'].dt.date
        cnts = df_logs['label'].value_counts()

        st.metric("Total Scans", len(df_logs))
        st.metric("Malware Detected", cnts.get('MALWARE', 0))

        st.subheader("Scan Trends Over Time")
        grp = df_logs.groupby(['date', 'label']).size().unstack(fill_value=0)
        fig1, ax1 = plt.subplots()
        grp.plot(ax=ax1, marker='o')
        ax1.set_xlabel('Date')
        ax1.set_ylabel('File Count')
        ax1.legend(title='Label')
        st.pyplot(fig1)
        st.write("Shows daily counts of legitimate vs malware scans for trend analysis.")

        st.subheader("Feature Trend Analysis")
        feat = st.selectbox("Select feature to track:", feature_list)
        fig2, ax2 = plt.subplots()
        ax2.plot(df_logs['datetime'], df_logs[feat], marker='x')
        ax2.set_xlabel('Time')
        ax2.set_ylabel(feat)
        st.pyplot(fig2)
        st.write("Tracks the selected feature value across scans to spot anomalies.")

# ───────────────────────────────────────────────────────────────────────────────
# Tab 4: Reports
# ───────────────────────────────────────────────────────────────────────────────
with tab_reports:
    st.header("📝 Automated Forensics Report")
    if st.button("Download PDF Report"):
        pdf = FPDF(); pdf.add_page(); pdf.set_font("Arial", 'B', 16)
        pdf.cell(0, 10, "Malware Analysis Report", ln=True)
        for ent in st.session_state["logs"]:
            pdf.set_font("Arial", size=12)
            line = f"File: {ent['file']} | Hash: {ent['hash'][:12]}... | Label: {ent['label']}"
            pdf.multi_cell(0, 8, line)
        path = "malware_report.pdf"; pdf.output(path)
        with open(path, 'rb') as f: st.download_button("Download PDF Report", f, file_name="malware_report.pdf")
